# 1. 初始化 Conda
source /home/uestc/miniconda3/etc/profile.d/conda.sh

# 2. 激活 StreamDSGN 环境
conda activate /data/jhb/conda_envs/streamdsgn4090

# 3. 进入项目
cd /data/jhb/workspace/streamDSGN

# 4. 指定 GPU
export CUDA_VISIBLE_DEVICES=0

# 5. 使用 Conda 环境里的 CUDA 12.8

export PATH=$CUDA_HOME/bin:$PATH

export LD_LIBRARY_PATH="$TORCH_LIB:$CUDA_HOME/lib:$CUDA_HOME/lib64:${LD_LIBRARY_PATH:-}"

# 7. RTX 4090 = SM89
export TORCH_CUDA_ARCH_LIST="8.9"

# 8. Torch extension/cache 路径
export TORCH_EXTENSIONS_DIR=/data/jhb/torch_extensions
export PIP_CACHE_DIR=/data/jhb/pip_cache
export CONDA_PKGS_DIRS=/data/jhb/conda_pkgs

# 9. StreamDSGN + 内置 mmdetection
export PYTHONPATH=/data/jhb/workspace/streamDSGN/mmdetection-v2.22.0:/data/jhb/workspace/streamDSGN:${PYTHONPATH:-}
